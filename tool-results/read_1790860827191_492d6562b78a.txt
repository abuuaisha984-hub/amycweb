    38→ */
    39→export function parseTranslations(raw: string | null | undefined): Record<string, any> {
    40→  if (!raw) return {}
    41→  try {
    42→    return JSON.parse(raw)
    43→  } catch {
    44→    return {}
    45→  }
    46→}

    48→/**
    49→ * Resolve a localized field value from an object's translations JSON.
    50→ * Falls back to the English default when the requested locale is missing.
    51→ */
    52→export function localizedField(
    53→  obj: { translations?: string | null } | null | undefined,
    54→  field: string,
    55→  locale: Locale,
    56→  fallback: string
    57→): string {
    58→  if (!obj) return fallback
    59→  if (locale === "en") return fallback
    60→  const tr = parseTranslations(obj.translations)
    61→  const loc = tr[locale]
    62→  if (loc && typeof loc === "object" && loc[field]) return loc[field]
    63→  return fallback
    64→}
    65→
    66→/**
    67→ * Check whether a given locale has a translation for a field.
    68→ */
    69→export function hasTranslation(
    70→  obj: { translations?: string | null } | null | undefined,
    71→  locale: Locale,
    72→  field?: string
    73→): boolean {
    74→  if (!obj || locale === "en") return true
    75→  const tr = parseTranslations(obj.translations)
    76→  const loc = tr[locale]
    77→  if (!loc || typeof loc !== "object") return false
    78→  if (field) return !!loc[field]
    79→  return Object.values(loc).some((v) => !!v)
    80→}

   812→/**
   813→ * Format a date according to locale.
   814→ */
   815→export function formatDate(date: Date | string, locale: Locale, opts?: Intl.DateTimeFormatOptions): string {
   816→  const d = typeof date === "string" ? new Date(date) : date
   817→  const loc = locale === "ar" ? "ar" : locale === "sw" ? "sw-TZ" : "en-GB"
   818→  try {
   819→    return new Intl.DateTimeFormat(loc, opts || { day: "numeric", month: "short", year: "numeric" }).format(d)
   820→  } catch {
   821→    return d.toLocaleDateString("en-GB", opts)
   822→  }
   823→}
   824→
   825→export function formatDateTime(date: Date | string, locale: Locale): string {
   826→  return formatDate(date, locale, { day: "numeric", month: "short", year: "numeric", hour: "2-digit", minute: "2-digit" })
   827→}

   829→// ── hreflang alternates helper for metadata ─────────────────────────────────
   830→export function buildAlternates(pathWithoutLocale: string): Record<string, string> {
   831→  const languages: Record<string, string> = {}
   832→  for (const loc of LOCALES) {
   833→    languages[loc] = `/${loc}${pathWithoutLocale}`
   834→  }
   835→  languages["x-default"] = `/en${pathWithoutLocale}`
   836→  return languages
   837→}
